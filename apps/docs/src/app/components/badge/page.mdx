import { Badge } from "@repo/vesper/badge";

export const storybookUrl = process.env.NODE_ENV === "development" ? "http://localhost:3000" : "/storybook";

# Badge

A small labeling component used to highlight status, categories, or metadata. Supports multiple color variants, sizes, and an optional leading icon. [View in Storybook]({`${storybookUrl}/badge`})

<Badge variant="accent">New</Badge>

## Options

| Prop | Type | Description | Default |
| --- | --- | --- | --- |
| `size` | `"sm" \| "md" \| "lg"` | The size of the badge. Affects padding and typography. | `"lg"` |
| `variant` | `"accent" \| "success" \| "warning" \| "danger" \| "info" \| "purple" \| "pink" \| "mint" \| "contrast"` | The color variant of the badge. | `"accent"` |
| `subtle` | `boolean` | When `true`, renders the badge with a more subdued appearance. | `false` |
| `icon` | `ReactNode` | An optional icon element rendered to the left of the badge text. | — |
| `as` | `ElementType` | The root element type for polymorphic rendering. | `"div"` |
| `children` | `ReactNode` | The text content of the badge. | — |

## Examples

```tsx
import { Badge } from "@repo/vesper/badge";

// Basic usage
<Badge>Default</Badge>
```

```tsx
// Color variants
<Badge variant="success">Active</Badge>
<Badge variant="danger">Error</Badge>
<Badge variant="warning">Pending</Badge>
<Badge variant="info">Info</Badge>
```

```tsx
// Sizes
<Badge size="sm">Small</Badge>
<Badge size="md">Medium</Badge>
<Badge size="lg">Large</Badge>
```

```tsx
// Subtle appearance
<Badge variant="success" subtle>Completed</Badge>
```

```tsx
// With icon
import { Checkmark } from "@repo/vesper/icons";

<Badge variant="success" icon={<Checkmark />}>Verified</Badge>
```

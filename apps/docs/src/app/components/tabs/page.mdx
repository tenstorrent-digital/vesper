import { Tabs } from "@tenstorrent/vesper/tabs";

import { STORYBOOK_URL } from "@/lib/constants";

# Tabs

A tabbed interface component for organizing content into switchable panels. Supports primary and secondary visual styles, icons in tab triggers, and controlled or uncontrolled selection. [View in Storybook]({`${STORYBOOK_URL}/tabs`})

<Tabs
  defaultValue="overview"
  items={[
    {
      label: "Overview",
      value: "overview",
      content: <p>Overview content goes here.</p>,
    },
    {
      label: "Settings",
      value: "settings",
      content: <p>Settings content goes here.</p>,
    },
    {
      label: "Activity",
      value: "activity",
      content: <p>Activity content goes here.</p>,
    },
  ]}
/>

## Options

| Prop             | Type                       | Description                                                                       | Default       |
| ---------------- | -------------------------- | --------------------------------------------------------------------------------- | ------------- |
| `variant`        | `"primary" \| "secondary"` | The visual style variant of the tabs.                                             | `"primary"`   |
| `items`          | `TabItem[]`                | The list of tab items. Each has `label`, `value`, optional `icon`, and `content`. | —             |
| `value`          | `string`                   | The value for the selected tab (controlled mode).                                 | —             |
| `defaultValue`   | `string`                   | The value of the tab to select by default (uncontrolled mode).                    | —             |
| `onValueChange`  | `(value: string) => void`  | Callback fired when a new tab is selected.                                        | —             |
| `activationMode` | `"automatic" \| "manual"`  | Whether a tab is activated on focus (`automatic`) or on click (`manual`).         | `"automatic"` |

### TabItem

| Prop      | Type        | Description                                                | Default |
| --------- | ----------- | ---------------------------------------------------------- | ------- |
| `label`   | `string`    | The text label displayed in the tab trigger.               | —       |
| `value`   | `string`    | A unique value identifying this tab.                       | —       |
| `icon`    | `ReactNode` | An optional icon element rendered before the label.        | —       |
| `content` | `ReactNode` | The content rendered in the panel when this tab is active. | —       |

## Examples

```tsx
import { Tabs } from "@tenstorrent/vesper/tabs";

// Basic usage
<Tabs
  defaultValue="tab1"
  items={[
    { label: "Tab 1", value: "tab1", content: <p>First panel</p> },
    { label: "Tab 2", value: "tab2", content: <p>Second panel</p> },
  ]}
/>;
```

```tsx
// Secondary variant
<Tabs
  variant="secondary"
  defaultValue="code"
  items={[
    { label: "Code", value: "code", content: <pre>...</pre> },
    { label: "Preview", value: "preview", content: <div>...</div> },
  ]}
/>
```

```tsx
// With icons
import { Gear, Globe } from "@tenstorrent/vesper/icons";

<Tabs
  defaultValue="general"
  items={[
    {
      label: "General",
      value: "general",
      icon: <Globe />,
      content: <p>General settings</p>,
    },
    {
      label: "Advanced",
      value: "advanced",
      icon: <Gear />,
      content: <p>Advanced settings</p>,
    },
  ]}
/>;
```

```tsx
// Controlled
const [tab, setTab] = useState("first");

<Tabs
  value={tab}
  onValueChange={setTab}
  items={[
    { label: "First", value: "first", content: <p>First</p> },
    { label: "Second", value: "second", content: <p>Second</p> },
  ]}
/>;
```

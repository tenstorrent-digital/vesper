# Menu

A dropdown menu component that displays a list of actionable items when triggered. Supports icons, descriptions, selected state, locked items, and disabled items. Built on Radix UI's dropdown menu primitive.

```tsx demo
import { Button } from "@tenstorrent/vesper/button";
import { Menu } from "@tenstorrent/vesper/menu";

export default function MenuDemo() {
  return (
    <Menu
      items={[
        { text: "Edit", onSelect: () => {} },
        { text: "Duplicate", onSelect: () => {} },
        { text: "Delete", style: "danger", onSelect: () => {} },
      ]}
    >
      <Button variant="subtle">Open Menu</Button>
    </Menu>
  );
}
```

## Options

| Prop           | Type                                                                                | Description                                                                                                      | Default    |
| -------------- | ----------------------------------------------------------------------------------- | ---------------------------------------------------------------------------------------------------------------- | ---------- |
| `items`        | `MenuItemProps[]`                                                                   | The list of menu items to render. Each item has `text`, optional `description`, `icon`, `style`, and `onSelect`. | —          |
| `children`     | `ReactElement`                                                                      | The trigger element that opens the menu.                                                                         | —          |
| `side`         | `"top" \| "bottom" \| "left" \| "right"`                                            | The preferred side to render the menu against.                                                                   | `"bottom"` |
| `sideOffset`   | `number`                                                                            | The distance in pixels from the trigger to the menu.                                                             | `8`        |
| `align`        | `"start" \| "center" \| "end"`                                                      | The alignment of the menu relative to the trigger.                                                               | `"start"`  |
| `alignOffset`  | `number`                                                                            | An offset in pixels from the aligned edge of the trigger.                                                        | —          |
| `width`        | `number`                                                                            | The width of the menu dropdown in pixels.                                                                        | `200`      |
| `open`         | `boolean`                                                                           | Controls the open state of the menu (controlled mode).                                                           | —          |
| `defaultOpen`  | `boolean`                                                                           | Whether the menu is open by default (uncontrolled mode).                                                         | `false`    |
| `onOpenChange` | `(open: boolean) => void`                                                           | Callback fired when the open state changes.                                                                      | —          |
| `container`    | `HTMLElement \| ShadowRoot \| null \| RefObject<HTMLElement \| ShadowRoot \| null>` | Specify the element or shadow root to portal the menu into.                                                      | –          |
| `anchor`       | `HTMLElement \| RefObject<HTMLElement \| null>`                                     | Specify the element to anchor the menu against. Default behavior anchors the menu to the triggering element      | –          |

### MenuItemProps

| Prop          | Type                                                            | Description                                                       | Default     |
| ------------- | --------------------------------------------------------------- | ----------------------------------------------------------------- | ----------- |
| `text`        | `string`                                                        | The text label displayed for the menu item.                       | —           |
| `description` | `string`                                                        | An optional secondary description displayed below the text label. | —           |
| `icon`        | `ReactNode`                                                     | An optional icon element rendered to the left of the text.        | —           |
| `style`       | `"default" \| "danger" \| "locked" \| "selected" \| "disabled"` | The visual and behavioral style of the menu item.                 | `"default"` |
| `onSelect`    | `() => void`                                                    | Callback fired when the menu item is selected.                    | —           |

## Examples

```tsx
import { Button } from "@tenstorrent/vesper/button";
import { Menu } from "@tenstorrent/vesper/menu";

// Basic menu
export default function BasicMenu() {
  return (
    <Menu
      items={[
        { text: "Copy", onSelect: () => console.log("Copy") },
        { text: "Paste", onSelect: () => console.log("Paste") },
        {
          text: "Delete",
          style: "danger",
          onSelect: () => console.log("Delete"),
        },
      ]}
    >
      <Button>Actions</Button>
    </Menu>
  );
}
```

```tsx
import { Button } from "@tenstorrent/vesper/button";
import { Copy, Gear, Trash } from "@tenstorrent/vesper/icons";
import { Menu } from "@tenstorrent/vesper/menu";

// With icons and descriptions
export default function MenuWithIconsAndDescriptions() {
  return (
    <Menu
      items={[
        { text: "Copy", icon: <Copy />, onSelect: () => console.log("Copy") },
        {
          text: "Settings",
          icon: <Gear />,
          description: "Manage preferences",
          onSelect: () => console.log("Settings"),
        },
        {
          text: "Delete",
          icon: <Trash />,
          style: "danger",
          onSelect: () => console.log("Delete"),
        },
      ]}
    >
      <Button>Actions</Button>
    </Menu>
  );
}
```

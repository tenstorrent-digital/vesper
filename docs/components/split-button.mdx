# Split Button

A compound button that combines a primary action button with a dropdown menu toggle. The primary button fires an action, while the secondary caret button opens a menu of alternative actions.

```tsx demo
import { SplitButton } from "@tenstorrent/vesper/split-button";

export default function SplitButtonDemo() {
  return (
    <SplitButton
      menuItems={[
        { text: "Save as draft", onSelect: () => {} },
        { text: "Save and publish", onSelect: () => {} },
      ]}
    >
      Save
    </SplitButton>
  );
}
```

## Options

| Prop                  | Type                                     | Description                                                     | Default         |
| --------------------- | ---------------------------------------- | --------------------------------------------------------------- | --------------- |
| `size`                | `"sm" \| "md" \| "lg"`                   | The size of both the action button and the menu toggle.         | `"md"`          |
| `variant`             | `"subtle" \| "contrast"`                 | The visual style variant applied to both buttons.               | `"contrast"`    |
| `menuItems`           | `MenuItemProps[]`                        | The list of menu items rendered in the dropdown.                | —               |
| `onClick`             | `MouseEventHandler<HTMLButtonElement>`   | Callback fired when the primary action button is clicked.       | —               |
| `menuButtonAriaLabel` | `string`                                 | An accessible `aria-label` for the menu toggle button.          | `"Toggle menu"` |
| `menuWidth`           | `number`                                 | The width of the dropdown menu in pixels.                       | `200`           |
| `menuSide`            | `"top" \| "bottom" \| "left" \| "right"` | The preferred side to render the dropdown menu.                 | `"bottom"`      |
| `menuSideOffset`      | `number`                                 | The distance in pixels from the button to the dropdown menu.    | `8`             |
| `menuAlign`           | `"start" \| "center" \| "end"`           | The alignment of the dropdown menu relative to the button.      | `"start"`       |
| `menuAlignOffset`     | `number`                                 | An offset in pixels from the aligned edge.                      | —               |
| `menuOpen`            | `boolean`                                | Controls the open state of the dropdown menu (controlled mode). | —               |
| `onMenuOpenChange`    | `(open: boolean) => void`                | Callback fired when the dropdown's open state changes.          | —               |
| `defaultMenuOpen`     | `boolean`                                | Whether the dropdown is open by default (uncontrolled mode).    | —               |
| `disabled`            | `boolean`                                | When `true`, disables both buttons.                             | `false`         |
| `children`            | `ReactNode`                              | The label content of the primary action button.                 | —               |

## Examples

```tsx
import { SplitButton } from "@tenstorrent/vesper/split-button";

// Basic usage
export default function BasicSplitButton() {
  return (
    <SplitButton
      onClick={() => console.log("Save")}
      menuItems={[
        { text: "Save as draft", onSelect: () => console.log("Save as draft") },
        {
          text: "Save and close",
          onSelect: () => console.log("Save and close"),
        },
      ]}
    >
      Save
    </SplitButton>
  );
}
```

```tsx
import { SplitButton } from "@tenstorrent/vesper/split-button";

// Subtle variant, small size
export default function SubtleSplitButton() {
  return (
    <SplitButton
      variant="subtle"
      size="sm"
      onClick={() => console.log("Action")}
      menuItems={[
        { text: "Option A", onSelect: () => {} },
        { text: "Option B", onSelect: () => {} },
      ]}
    >
      Actions
    </SplitButton>
  );
}
```

```tsx
import { SplitButton } from "@tenstorrent/vesper/split-button";

// Disabled
export default function DisabledSplitButton() {
  return (
    <SplitButton disabled menuItems={[]}>
      Disabled
    </SplitButton>
  );
}
```
